Source code

import { htmlpdf } from 'htmlpdfx.js';

const blob = await htmlpdf(element, {
  format: 'a4',
  margin: 10,
  compress: true,

  // Custom font via URL
  fonts: [{
    fontFamily: 'Roboto',
    fontUrl: '...roboto-regular.ttf',
    isDefault: true,
  }],

  // Header / footer
  header: {
    height: 10,
    render(doc, info) {
      doc.setFontSize(9);
      doc.text('Annual Report 2024',
        info.margin, info.margin - 2);
    },
  },
  footer: {
    height: 10,
    render(doc, info) {
      doc.setFontSize(8);
      doc.text(
        `Page ${info.pageNumber} / ${info.totalPages}`,
        info.pageWidth / 2,
        info.pageHeight - info.margin + 4,
        { align: 'center' });
    },
  },

  // Repeat table header
  tables: [{
    selector: '#metrics-table',
    repeatHeader: 'thead',
    pageBreakBorder: '1px solid #bbb',
  }],

  // Progress callback
  onProgress({ stage, progress }) {
    updateProgressBar(stage, progress);
  },
});

// Preview in iframe
iframe.src = URL.createObjectURL(blob);

Pipeline stages
  • Clone & measure — DOM is cloned into a hidden iframe; computed styles and bounding boxes are captured.
  • Image preload — All images and canvases are rasterized before the iframe is destroyed.
  • Font load & register — Custom fonts are fetched, Base64-encoded, and injected into jsPDF.
  • Paginate → Render → Output — Single-pass pagination builds allPlacements; each node is drawn; the PDF blob is returned.
